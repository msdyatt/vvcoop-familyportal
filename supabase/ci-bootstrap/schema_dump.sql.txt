select line from (
  select format('POLICY|%s|%s|%s|%s|%s', tablename, policyname, cmd, array_to_string(roles, ','), md5(coalesce(qual, '') || '#' || coalesce(with_check, ''))) as line
  from pg_policies where schemaname = 'public'
  union all
  select format('FUNC|%s.%s(%s)|%s|%s|%s', n.nspname, p.proname, pg_get_function_identity_arguments(p.oid), p.prosecdef, coalesce(array_to_string(p.proconfig, ','), ''), md5(p.prosrc))
  from pg_proc p join pg_namespace n on n.oid = p.pronamespace
  where n.nspname in ('public', 'private') and p.prokind = 'f'
  union all
  select format('TRIGGER|%s|%s|%s', c.relname, t.tgname, md5(pg_get_triggerdef(t.oid)))
  from pg_trigger t join pg_class c on c.oid = t.tgrelid join pg_namespace n on n.oid = c.relnamespace
  where n.nspname = 'public' and not t.tgisinternal
  union all
  select format('RLS|%s|%s', c.relname, c.relrowsecurity)
  from pg_class c join pg_namespace n on n.oid = c.relnamespace where n.nspname = 'public' and c.relkind = 'r'
) x order by line;
